;; =========================================================================
;; FILE: modular-sim/wasm-core/src/engine.wat
;; DESC: atomic-nand execution kernel
;; =========================================================================

(module
  ;; grab the shared memory from the js side
  (import "env" "memory" (memory 1))

  ;; start of node states (region a)
  (global $REGION_A_BASE i32 (i32.const 0))
  ;; start of instructions (region b)
  (global $REGION_B_BASE i32 (i32.const 16384))

  ;; -----------------------------------------------------------------------
  ;; ATOMIC PRIMITIVE: $nand
  ;; -----------------------------------------------------------------------
  (func $nand (param $a i32) (param $b i32) (result i32)
    local.get $a      ;; push a
    local.get $b      ;; push b
    i32.and           ;; logic and
    i32.const 1       ;; push mask
    i32.xor           ;; flip it for nand
  )

  ;; -----------------------------------------------------------------------
  ;; CORE KERNEL: $tick
  ;;
  ;; how this works (pseudocode):
  ;; for each instruction (i) up to instruction_count:
  ;;   1. ptr = (i * 16) + 16384             ;; figure out where we are in region b
  ;;   2. a_idx = read(ptr + 4)              ;; load up input a state
  ;;      val_a = read(a_idx * 4)
  ;;   3. b_idx = read(ptr + 8)              ;; load up input b state
  ;;      val_b = read(b_idx * 4)
  ;;   4. target_addr = read(ptr) * 4        ;; get the target address
  ;;   5. opcode = read(ptr + 12)            ;; switch statement for the opcodes
  ;;      case 0: result = nand(a, b)        ;; op 0: basic nand
  ;;      case 2: ...                        ;; op 2: clock counter logic
  ;;      case 5: result = nand(a, a)        ;; op 5: not gate built with nand
  ;;      case 6: ...                        ;; op 6: and gate
  ;;      ... etc
  ;;   6. write(target_addr, result)         ;; save the result back to memory
  ;; -----------------------------------------------------------------------
  (func $tick (param $instruction_count i32)
    (local $i i32)            ;; loop index
    (local $ptr i32)          ;; address for current inst
    (local $val_a i32)        ;; temp input a
    (local $val_b i32)        ;; temp input b
    (local $target_addr i32)  ;; where we save the result
    (local $opcode i32)       ;; what we're doing
    
    i32.const 0
    local.set $i              ;; start at zero

    (loop $eval_loop
      local.get $i
      local.get $instruction_count
      i32.lt_s                ;; keep going while i < count
      (if
        (then
          ;; figure out where we are in region b
          local.get $i
          i32.const 4         ;; bitshift left 4 is mul by 16
          i32.shl
          global.get $REGION_B_BASE
          i32.add
          local.set $ptr      ;; save the address

          ;; load up input a state
          local.get $ptr i32.load offset=4 i32.const 2 i32.shl i32.load
          local.set $val_a

          ;; load up input b state
          local.get $ptr i32.load offset=8 i32.const 2 i32.shl i32.load
          local.set $val_b

          ;; get the target address and leave it on stack for the store at the end
          local.get $ptr i32.load i32.const 2 i32.shl
          local.tee $target_addr

          ;; load the opcode
          local.get $ptr i32.load offset=12
          local.set $opcode

          ;; giant if/else chain replacing br_table
          local.get $opcode i32.const 0 i32.eq
          (if (result i32)
            (then local.get $val_a local.get $val_b call $nand) ;; op 0: nand
            (else
              local.get $opcode i32.const 1 i32.eq
              (if (result i32)
                (then local.get $val_a) ;; op 1: dff
                (else
                  local.get $opcode i32.const 2 i32.eq
                  (if (result i32)
                    (then 
                      ;; op 2: clock
                      local.get $val_a i32.const 1 i32.add local.tee $val_a
                      local.get $val_b i32.ge_s
                      (if (result i32)
                        (then 
                          local.get $ptr i32.load offset=4 i32.const 2 i32.shl i32.const 0 i32.store
                          local.get $target_addr i32.load i32.const 1 i32.xor
                        )
                        (else 
                          local.get $ptr i32.load offset=4 i32.const 2 i32.shl local.get $val_a i32.store
                          local.get $target_addr i32.load
                        )
                      )
                    )
                    (else
                      local.get $opcode i32.const 3 i32.eq
                      (if (result i32)
                        (then 
                          ;; op 3: tristate
                          local.get $val_b i32.const 1 i32.eq (if (result i32) (then local.get $val_a) (else i32.const 2))
                        )
                        (else
                          local.get $opcode i32.const 4 i32.eq
                          (if (result i32)
                            (then 
                              ;; op 4: tff
                              local.get $val_b i32.const 1 i32.eq (if (result i32) (then local.get $val_a i32.const 1 i32.xor) (else local.get $val_a))
                            )
                            (else
                              local.get $opcode i32.const 5 i32.eq
                              (if (result i32)
                                (then 
                                  ;; op 5: not
                                  local.get $val_a local.get $val_a call $nand
                                )
                                (else
                                  local.get $opcode i32.const 6 i32.eq
                                  (if (result i32)
                                    (then 
                                      ;; op 6: and
                                      local.get $val_a local.get $val_b call $nand local.tee $val_a local.get $val_a call $nand
                                    )
                                    (else
                                      local.get $opcode i32.const 7 i32.eq
                                      (if (result i32)
                                        (then 
                                          ;; op 7: or
                                          local.get $val_a local.get $val_a call $nand
                                          local.get $val_b local.get $val_b call $nand
                                          call $nand
                                        )
                                        (else
                                          local.get $opcode i32.const 8 i32.eq
                                          (if (result i32)
                                            (then 
                                              ;; op 8: nor
                                              local.get $val_a local.get $val_a call $nand local.set $val_a
                                              local.get $val_b local.get $val_b call $nand local.get $val_a call $nand
                                              local.tee $val_a local.get $val_a call $nand
                                            )
                                            (else
                                              local.get $opcode i32.const 9 i32.eq
                                              (if (result i32)
                                                (then 
                                                  ;; op 9: xor
                                                  local.get $val_a local.get $val_b call $nand local.set $opcode
                                                  local.get $val_a local.get $opcode call $nand
                                                  local.get $val_b local.get $opcode call $nand
                                                  call $nand
                                                )
                                                (else
                                                  local.get $opcode i32.const 10 i32.eq
                                                  (if (result i32)
                                                    (then 
                                                      ;; op 10: xnor
                                                      local.get $val_a local.get $val_b call $nand local.set $opcode
                                                      local.get $val_a local.get $opcode call $nand
                                                      local.get $val_b local.get $opcode call $nand
                                                      call $nand
                                                      local.tee $val_a local.get $val_a call $nand
                                                    )
                                                    (else
                                                      local.get $opcode i32.const 11 i32.eq
                                                      (if (result i32)
                                                        (then 
                                                          ;; op 11: bus resolve
                                                          local.get $val_a i32.const 2 i32.eq (if (result i32) (then local.get $val_b) (else local.get $val_a))
                                                        )
                                                        (else
                                                          ;; fallback / default
                                                          local.get $val_a local.get $val_b call $nand
                                                        )
                                                      )
                                                    )
                                                  )
                                                )
                                              )
                                            )
                                          )
                                        )
                                      )
                                    )
                                  )
                                )
                              )
                            )
                          )
                        )
                      )
                    )
                  )
                )
              )
            )
          )

          ;; save the result back to memory
          i32.store

          ;; onto the next instruction
          local.get $i i32.const 1 i32.add local.set $i
          br $eval_loop
        )
      )
    )
  )

  ;; let js call the tick function
  (export "tick" (func $tick))
)
